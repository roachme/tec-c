# Initialize tec
tec init

# We create a testing environment, name it 'test' and add some tasks
tec env add test
tec add test1
tec add test2
tec add test3

# Let's see those tasks
tec ls

# Switch to the user's home directory
cd

# Switch back to the current task directory
tec cd

# Switch to previous task
tec cd -

# Delete the current task
tec rm -f

# The user's current directory is automatically switched to the previous one
pwd

# The current task directory is removed
tec ls
